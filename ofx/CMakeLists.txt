cmake_minimum_required(VERSION 3.21)
project(AdvancedDirtRemovalOFX VERSION 2.0.0 LANGUAGES C CXX)
# Self-contained OFX plugin with the engine linked in statically.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Self-contained binaries: no MSVC redistributable or newer libstdc++ needed.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures" FORCE)
endif()
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS" FORCE)
endif()
if(MSVC)
  add_compile_definitions(_DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
endif()
if(APPLE)
  set(OFX_ARCH MacOS)
elseif(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(OFX_ARCH Win64)
elseif(UNIX AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(OFX_ARCH Linux-x86-64)
else()
  message(FATAL_ERROR "Supported targets: macOS, Windows x64, Linux x64")
endif()

if(NOT TARGET adr_engine_static)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../engine ${CMAKE_BINARY_DIR}/engine)
endif()

set(ADR_BUNDLE "AdvancedDirtRemoval.ofx.bundle")
set(OFX_CONTENTS "${CMAKE_BINARY_DIR}/${ADR_BUNDLE}/Contents")
add_library(AdvancedDirtRemoval MODULE src/adr_ofx.cpp)
target_link_libraries(AdvancedDirtRemoval PRIVATE adr_engine_static)
target_include_directories(AdvancedDirtRemoval PRIVATE vendor/openfx/include)

# -DADR_SOURCE_REVISION=<id> when building outside a git checkout; otherwise git describe.
find_package(Git QUIET)
set(ADR_SOURCE_REVISION "" CACHE STRING "Source revision for the diagnostic log (default: git describe)")
set(ADR_REVISION "${ADR_SOURCE_REVISION}")
if(NOT ADR_REVISION AND GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} -C ${CMAKE_CURRENT_SOURCE_DIR} describe --always --dirty
    OUTPUT_VARIABLE ADR_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE ADR_GIT_RESULT)
  if(NOT ADR_GIT_RESULT EQUAL 0)
    set(ADR_REVISION "")
  endif()
endif()
if(NOT ADR_REVISION)
  set(ADR_REVISION "unknown")
endif()
set(ADR_VERSION_DEFS ADR_PLUGIN_VERSION="${PROJECT_VERSION}" ADR_SOURCE_REVISION="${ADR_REVISION}")
target_compile_definitions(AdvancedDirtRemoval PRIVATE ${ADR_VERSION_DEFS})
set_target_properties(AdvancedDirtRemoval PROPERTIES PREFIX "" SUFFIX ".ofx"
  LIBRARY_OUTPUT_DIRECTORY "${OFX_CONTENTS}/${OFX_ARCH}" RUNTIME_OUTPUT_DIRECTORY "${OFX_CONTENTS}/${OFX_ARCH}"
  CXX_VISIBILITY_PRESET hidden)
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(AdvancedDirtRemoval PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${OFX_CONTENTS}/${OFX_ARCH}"
    RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${OFX_CONTENTS}/${OFX_ARCH}")
endforeach()
if(MSVC)
  target_compile_options(AdvancedDirtRemoval PRIVATE /W4 /EHsc)
else()
  target_compile_options(AdvancedDirtRemoval PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  target_link_libraries(AdvancedDirtRemoval PRIVATE ${CMAKE_DL_LIBS})
  if(UNIX AND NOT APPLE)
    target_link_options(AdvancedDirtRemoval PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL)
  endif()
endif()

file(MAKE_DIRECTORY "${OFX_CONTENTS}/Resources")
configure_file(Info.plist.in "${OFX_CONTENTS}/Info.plist" @ONLY)
configure_file(vendor/openfx/LICENSE.md "${OFX_CONTENTS}/Resources/OpenFX-LICENSE.md" COPYONLY)
configure_file(vendor/openfx/UPSTREAM.json "${OFX_CONTENTS}/Resources/OpenFX-UPSTREAM.json" COPYONLY)
configure_file(../LICENSE "${OFX_CONTENTS}/Resources/LICENSE.txt" COPYONLY)
if(APPLE)
  # Ad-hoc sign the whole bundle (both architectures); hosts on Apple silicon
  # refuse unsigned code. Pass a Developer ID identity for distribution.
  set(ADR_CODESIGN_IDENTITY "-" CACHE STRING "codesign identity for the OFX bundle")
  set_property(TARGET AdvancedDirtRemoval APPEND PROPERTY LINK_DEPENDS
    "${OFX_CONTENTS}/Info.plist" "${OFX_CONTENTS}/Resources/LICENSE.txt")
  add_custom_command(TARGET AdvancedDirtRemoval POST_BUILD
    COMMAND codesign --force --timestamp=none --sign "${ADR_CODESIGN_IDENTITY}" "${CMAKE_BINARY_DIR}/${ADR_BUNDLE}"
    COMMENT "Signing ${ADR_BUNDLE}" VERBATIM)
endif()

option(ADR_BUILD_TESTS "Build the OFX host contract test" ON)
if(ADR_BUILD_TESTS)
  enable_testing()
  add_executable(adr_ofx_contract tests/contract_host.cpp)
  target_include_directories(adr_ofx_contract PRIVATE vendor/openfx/include)
  target_link_libraries(adr_ofx_contract PRIVATE adr_engine_static)
  target_compile_definitions(adr_ofx_contract PRIVATE ${ADR_VERSION_DEFS})
  if(NOT MSVC)
    target_compile_options(adr_ofx_contract PRIVATE -ffp-contract=off)
  endif()
endif()
